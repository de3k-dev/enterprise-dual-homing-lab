!=================================================
! Hostname & Base Configuration
!=================================================
hostname HQ-R1-GW
ip cef

!=================================================
! Interfaces
!=================================================
interface Loopback1
 ip address 1.1.1.1 255.255.255.255
 no shutdown

! Uplink to ISP-1 (eBGP)
interface Ethernet0/0
 ip address 198.51.100.2 255.255.255.252
 ip nat outside
 no shutdown

! Interconnect to Core-SW1 (OSPF + BFD)
interface Ethernet0/1
 ip address 10.10.11.1 255.255.255.252
 ip nat inside
 bfd interval 50 min_rx 50 multiplier 3
 no shutdown

! Link to HQ-R2-GW (iBGP + OSPF + BFD)
interface Ethernet0/2
 ip address 10.0.0.1 255.255.255.252
 ip nat inside
 bfd interval 50 min_rx 50 multiplier 3
 no shutdown

!=================================================
! Dynamic Routing: OSPFv2
!=================================================
router ospf 1
 router-id 1.1.1.1
 auto-cost reference-bandwidth 100000
 network 1.1.1.1 0.0.0.0 area 0
 network 10.0.0.0 0.0.0.3 area 0
 network 10.10.11.0 0.0.0.3 area 0
 bfd all-interfaces

!=================================================
! Dynamic Routing: BGP (AS 65001)
!=================================================
router bgp 65001
 bgp log-neighbor-changes
 network 172.16.0.0
 ! iBGP Neighbor (HQ-R2-GW)
 neighbor 2.2.2.2 remote-as 65001
 neighbor 2.2.2.2 update-source Loopback1
 neighbor 2.2.2.2 next-hop-self
 ! eBGP Neighbor (ISP-1)
 neighbor 198.51.100.1 remote-as 100
 neighbor 198.51.100.1 route-map LOCAL_PREF in

! Static summary route to Null0 for BGP advertisement
ip route 172.16.0.0 255.255.0.0 Null0

!=================================================
! Route-Maps & BGP Traffic Engineering
!=================================================
route-map LOCAL_PREF permit 10
 set local-preference 200

!=================================================
! NAT & ACL Setup
!=================================================
access-list 1 permit 172.16.0.0 0.0.255.255
ip nat inside source list 1 interface Ethernet0/0 overload

line con 0
 logging synchronous
